-- ============================================================
-- GO-TANI: SUPABASE POSTGRESQL COMPLETE SCHEMA & POLICIES
-- ============================================================

-- 1. TABEL: bibit_tanaman (30 Data Bibit Unggul)
CREATE TABLE IF NOT EXISTS public.bibit_tanaman (
  id BIGSERIAL PRIMARY KEY,
  nama_bibit TEXT NOT NULL,
  kategori TEXT NOT NULL,
  musim_tanam TEXT,
  masa_panen TEXT,
  harga_estimasi NUMERIC DEFAULT 0,
  stok BOOLEAN DEFAULT true,
  gambar_url TEXT,
  deskripsi TEXT,
  cara_tanam TEXT,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- RLS Bibit
ALTER TABLE public.bibit_tanaman ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read bibit" ON public.bibit_tanaman FOR SELECT USING (true);
CREATE POLICY "Admin CRUD bibit" ON public.bibit_tanaman FOR ALL USING (auth.role() = 'authenticated');

-- 2. TABEL: pesan_kontak (Formulir Pengunjung)
CREATE TABLE IF NOT EXISTS public.pesan_kontak (
  id BIGSERIAL PRIMARY KEY,
  nama TEXT NOT NULL,
  email TEXT NOT NULL,
  subjek TEXT,
  pesan TEXT NOT NULL,
  is_read BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- RLS Pesan
ALTER TABLE public.pesan_kontak ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public insert pesan" ON public.pesan_kontak FOR INSERT WITH CHECK (true);
CREATE POLICY "Admin read pesan" ON public.pesan_kontak FOR ALL USING (auth.role() = 'authenticated');
