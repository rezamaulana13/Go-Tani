-- ============================================================
-- GO-TANI: COMPLETE SUPABASE POSTGRESQL SCHEMA & POLICIES
-- ============================================================

-- 1. TABEL: bibit_tanaman
CREATE TABLE IF NOT EXISTS public.bibit_tanaman (
  id BIGSERIAL PRIMARY KEY,
  nama_bibit TEXT NOT NULL,
  kategori TEXT NOT NULL,
  musim_tanam TEXT,
  masa_panen TEXT,
  harga_estimasi NUMERIC DEFAULT 0,
  stok BOOLEAN DEFAULT true,
  gambar_url TEXT,
  deskripsi TEXT,
  cara_tanam TEXT,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- RLS bibit_tanaman
ALTER TABLE public.bibit_tanaman ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read bibit" ON public.bibit_tanaman FOR SELECT USING (true);
CREATE POLICY "Admin CRUD bibit" ON public.bibit_tanaman FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

-- 2. TABEL: pesan_kontak
CREATE TABLE IF NOT EXISTS public.pesan_kontak (
  id BIGSERIAL PRIMARY KEY,
  nama TEXT NOT NULL,
  email TEXT NOT NULL,
  subjek TEXT,
  pesan TEXT NOT NULL,
  is_read BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- RLS pesan_kontak
ALTER TABLE public.pesan_kontak ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public insert pesan" ON public.pesan_kontak FOR INSERT TO anon, authenticated WITH CHECK (true);
CREATE POLICY "Admin CRUD pesan" ON public.pesan_kontak FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

-- 3. TABEL: blog_posts
CREATE TABLE IF NOT EXISTS public.blog_posts (
  id BIGSERIAL PRIMARY KEY,
  judul TEXT NOT NULL,
  slug TEXT NOT NULL UNIQUE,
  kategori TEXT DEFAULT 'Umum',
  author TEXT DEFAULT 'Tim Go-Tani',
  excerpt TEXT NOT NULL,
  konten TEXT,
  thumbnail_url TEXT,
  is_published BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- RLS blog_posts
ALTER TABLE public.blog_posts ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read blog" ON public.blog_posts FOR SELECT USING (true);
CREATE POLICY "Admin CRUD blog" ON public.blog_posts FOR ALL USING (auth.role() = 'authenticated');
